    GLuint CompileMapShader(GLenum type, const char* source)
    {
        GLuint shader = glCreateShader(type);
        glShaderSource(shader, 1, &source, nullptr);
        glCompileShader(shader);
        return shader;
    }

    void InitCityShadows()
    {
        const char* vs = R"(
#version 330 core
layout(location=0) in vec3 aPos;
uniform mat4 model;
uniform mat4 lightSpaceMatrix;
void main() { gl_Position = lightSpaceMatrix * model * vec4(aPos, 1.0); }
)";
        const char* fs = R"(
#version 330 core
void main() {}
)";
        GLuint vertex = CompileMapShader(GL_VERTEX_SHADER, vs);
        GLuint fragment = CompileMapShader(GL_FRAGMENT_SHADER, fs);
        cityShadowShader = glCreateProgram();
        glAttachShader(cityShadowShader, vertex);
        glAttachShader(cityShadowShader, fragment);
        glLinkProgram(cityShadowShader);
        glDeleteShader(vertex);
        glDeleteShader(fragment);

        glGenFramebuffers(1, &cityShadowFbo);
        glGenTextures(1, &cityShadowDepth);
        glBindTexture(GL_TEXTURE_2D, cityShadowDepth);
        glTexImage2D(GL_TEXTURE_2D, 0, GL_DEPTH_COMPONENT24,
                     CITY_SHADOW_SIZE, CITY_SHADOW_SIZE, 0,
                     GL_DEPTH_COMPONENT, GL_FLOAT, nullptr);
        glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
        glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
        glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_BORDER);
        glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_BORDER);
        const float border[] = { 1, 1, 1, 1 };
        glTexParameterfv(GL_TEXTURE_2D, GL_TEXTURE_BORDER_COLOR, border);
        glBindFramebuffer(GL_FRAMEBUFFER, cityShadowFbo);
        glFramebufferTexture2D(GL_FRAMEBUFFER, GL_DEPTH_ATTACHMENT,
                               GL_TEXTURE_2D, cityShadowDepth, 0);
        glDrawBuffer(GL_NONE);
        glReadBuffer(GL_NONE);
        glBindFramebuffer(GL_FRAMEBUFFER, 0);
    }

    bool CircleHitsCity(const glm::vec3& worldPosition, float radius)
    {
        return drownedCityModel.collidesCylinder(worldPosition, radius, 1.8f);
    }

    float PathDistance(float x, float z)
    {
        float d1 = std::abs(x);
        float curve1Z = std::sin(x * 0.06f) * 8.0f;
        float d2 = std::abs(z - curve1Z);
        float d3 = std::abs(z - 7.0f);
        float d4 = std::abs(x);
        float cross = std::min(d3, d4);
        return std::min({ d1, d2, cross });
    }

    float RawTerrainHeight(float x, float z)
    {
        float hills = std::sin(x * 0.05f) * std::cos(z * 0.04f) * 0.75f
            + std::sin(x * 0.12f + 1.3f) * std::cos(z * 0.08f - 0.7f) * 0.35f
            + std::sin(x * 0.03f - 0.5f) * std::sin(z * 0.025f + 1.0f) * 0.85f;

        float dist = std::sqrt(x * x + z * z);
        float centerFlat = 1.0f - std::min(dist / 12.0f, 1.0f);
        centerFlat = centerFlat * centerFlat;
        hills *= (1.0f - centerFlat * 0.9f);

        float pd = PathDistance(x, z);
        float pathBlend = 1.0f - std::min(pd / 2.8f, 1.0f);
        pathBlend = pathBlend * pathBlend * (3.0f - 2.0f * pathBlend);
        hills *= (1.0f - pathBlend * 0.9f);

        return hills;
    }
}

float GetTerrainHeight(float x, float z)
{
    if (drownedCityLoaded && x >= -18.0f && x <= 18.0f && z >= -15.0f && z <= 18.0f)
        return 0.64f;
    return RawTerrainHeight(x, z);
}

glm::vec3 ResolveMapCollision(const glm::vec3& from, const glm::vec3& desired, float radius)
{
    if (!drownedCityLoaded) return desired;
    glm::vec3 result = from;
    glm::vec3 testX(desired.x, desired.y, from.z);
    if (!CircleHitsCity(testX, radius)) result.x = desired.x;
    glm::vec3 testZ(result.x, desired.y, desired.z);
    if (!CircleHitsCity(testZ, radius)) result.z = desired.z;
    result.y = desired.y;
    return result;
}

float GetMapMoveSpeedMultiplier(const glm::vec3& position)
{
    return GetMapSurfaceProperties(position).speedMultiplier;
}

MapSurfaceProperties GetMapSurfaceProperties(const glm::vec3& position)
{
    if (!drownedCityLoaded) return { MapSurfaceType::Soil, 1.0f, 0.78f, 0.12f };
    const bool inCanal = std::abs(position.x) < 4.75f &&
                         position.z > -15.0f && position.z < 18.0f;
    if (inCanal)
        return { MapSurfaceType::ShallowWater, 0.68f, 0.42f, 0.48f };
    const bool onDistrict = position.x >= -18.0f && position.x <= 18.0f &&
                            position.z >= -15.0f && position.z <= 18.0f;
    if (onDistrict)
        return { MapSurfaceType::WetAsphalt, 0.92f, 0.67f, 0.16f };
    return { MapSurfaceType::Soil, 1.0f, 0.78f, 0.12f };
}

void InitSimpleMap()
{
    if (initialized) return;
    initialized = true;

    const char* cityPaths[] = {
        "Resource/DrownedCity/Exports/ENV_FC01_DrownedTransitDistrict.glb",
        "../trnsei/Resource/DrownedCity/Exports/ENV_FC01_DrownedTransitDistrict.glb",
        "../../trnsei/Resource/DrownedCity/Exports/ENV_FC01_DrownedTransitDistrict.glb"
    };
    for (const char* path : cityPaths) {
        if (drownedCityModel.load(path, false)) {
            drownedCityLoaded = true;
            break;
        }
    }
    InitCityShadows();

    const float groundSize = 60.0f;
    const int gridRes = 60;

    std::vector<float> groundVerts;
    for (int z = 0; z <= gridRes; ++z) {
        for (int x = 0; x <= gridRes; ++x) {
            float fx = ((float)x / gridRes - 0.5f) * groundSize * 2.0f;
            float fz = ((float)z / gridRes - 0.5f) * groundSize * 2.0f;
            float height = RawTerrainHeight(fx, fz);

            float pd = PathDistance(fx, fz);
            float pathBlend = 1.0f - std::min(pd / 2.8f, 1.0f);
            pathBlend = pathBlend * pathBlend * (3.0f - 2.0f * pathBlend);

            float heightFactor = std::max(0.0f, std::min(height / 4.0f, 1.0f));
            glm::vec3 lowGrass(0.035f, 0.19f, 0.23f);
            glm::vec3 highGrass(0.10f, 0.31f, 0.32f);
            glm::vec3 grassColor = glm::mix(lowGrass, highGrass, heightFactor);
            glm::vec3 pathColor(0.34f, 0.42f, 0.42f);
            glm::vec3 edgeColor(0.13f, 0.28f, 0.30f);

            glm::vec3 color;
            if (pathBlend > 0.6f)
                color = glm::mix(edgeColor, pathColor, (pathBlend - 0.6f) / 0.4f);
            else if (pathBlend > 0.1f)
                color = glm::mix(grassColor, edgeColor, (pathBlend - 0.1f) / 0.5f);
            else {
                float noise = std::sin(fx * 0.5f + fz * 0.3f) * 0.03f;
                color = grassColor + glm::vec3(noise, noise * 0.5f, -noise);
            }

            groundVerts.insert(groundVerts.end(), {
                fx, height, fz, 0.0f, 1.0f, 0.0f, color.r, color.g, color.b
            });
        }
    }

    std::vector<unsigned int> groundIdx;
    for (int z = 0; z < gridRes; ++z) {
        for (int x = 0; x < gridRes; ++x) {
            unsigned int tl = z * (gridRes + 1) + x;
            unsigned int tr = tl + 1;
            unsigned int bl = (z + 1) * (gridRes + 1) + x;
            unsigned int br = bl + 1;
            groundIdx.insert(groundIdx.end(), { tl, bl, tr, tr, bl, br });
        }
    }
    groundIndexCount = (int)groundIdx.size();

    GLuint groundEbo = 0;
    glGenVertexArrays(1, &mapGroundVao);
    glGenBuffers(1, &mapGroundVbo);
    glGenBuffers(1, &groundEbo);
    glBindVertexArray(mapGroundVao);
    glBindBuffer(GL_ARRAY_BUFFER, mapGroundVbo);
    glBufferData(GL_ARRAY_BUFFER, groundVerts.size() * sizeof(float), groundVerts.data(), GL_STATIC_DRAW);
    glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, groundEbo);
    glBufferData(GL_ELEMENT_ARRAY_BUFFER, groundIdx.size() * sizeof(unsigned int), groundIdx.data(), GL_STATIC_DRAW);
    glVertexAttribPointer(0, 3, GL_FLOAT, GL_FALSE, 9 * sizeof(float), nullptr);
    glEnableVertexAttribArray(0);
    glVertexAttribPointer(1, 3, GL_FLOAT, GL_FALSE, 9 * sizeof(float), (void*)(3 * sizeof(float)));
    glEnableVertexAttribArray(1);
    glVertexAttribPointer(2, 3, GL_FLOAT, GL_FALSE, 9 * sizeof(float), (void*)(6 * sizeof(float)));
    glEnableVertexAttribArray(2);
    glBindVertexArray(0);

    const float cubeV[] = {
        -0.5f,0,-0.5f, 0,0,-1, 0.5f,0,-0.5f, 0,0,-1, 0.5f,1,-0.5f, 0,0,-1, -0.5f,1,-0.5f, 0,0,-1,
        -0.5f,0,0.5f, 0,0,1, 0.5f,0,0.5f, 0,0,1, 0.5f,1,0.5f, 0,0,1, -0.5f,1,0.5f, 0,0,1,
        -0.5f,0,-0.5f, -1,0,0, -0.5f,0,0.5f, -1,0,0, -0.5f,1,0.5f, -1,0,0, -0.5f,1,-0.5f, -1,0,0,
        0.5f,0,-0.5f, 1,0,0, 0.5f,0,0.5f, 1,0,0, 0.5f,1,0.5f, 1,0,0, 0.5f,1,-0.5f, 1,0,0,
        -0.5f,1,-0.5f, 0,1,0, 0.5f,1,-0.5f, 0,1,0, 0.5f,1,0.5f, 0,1,0, -0.5f,1,0.5f, 0,1,0,
    };
    const unsigned int cubeI[] = {
        0,1,2,2,3,0, 4,6,5,6,4,7, 8,9,10,10,11,8,
        12,15,14,14,13,12, 16,17,18,18,19,16
    };
    glGenVertexArrays(1, &mapPropVao);
    glGenBuffers(1, &mapPropVbo);
    glGenBuffers(1, &mapPropEbo);
    glBindVertexArray(mapPropVao);
    glBindBuffer(GL_ARRAY_BUFFER, mapPropVbo);
    glBufferData(GL_ARRAY_BUFFER, sizeof(cubeV), cubeV, GL_STATIC_DRAW);
    glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, mapPropEbo);
    glBufferData(GL_ELEMENT_ARRAY_BUFFER, sizeof(cubeI), cubeI, GL_STATIC_DRAW);
    glVertexAttribPointer(0, 3, GL_FLOAT, GL_FALSE, 6 * sizeof(float), nullptr);
    glEnableVertexAttribArray(0);
    glVertexAttribPointer(1, 3, GL_FLOAT, GL_FALSE, 6 * sizeof(float), (void*)(3 * sizeof(float)));
    glEnableVertexAttribArray(1);
    glBindVertexArray(0);
    mapPropIndexCount = 30;

    srand(42);

    auto addProp = [](float x, float z, glm::vec3 s, glm::vec3 c) {
        mapProps.push_back({ glm::vec3(x, RawTerrainHeight(x, z), z), s, c });
    };

    for (int i = 0; i < 25; ++i) {
        float x = ((rand() % 1000) / 1000.0f - 0.5f) * 90.0f;
        float z = ((rand() % 1000) / 1000.0f - 0.5f) * 90.0f;
        if (std::sqrt(x*x + z*z) < 5.0f || PathDistance(x, z) < 4.5f) continue;
        float h = 3.0f + (rand() % 100) / 50.0f;
        float groundY = RawTerrainHeight(x, z);
        addProp(x, z, glm::vec3(0.38f, h, 0.38f), glm::vec3(0.16f, 0.24f, 0.25f));
        float r = 1.8f + (rand() % 100) / 80.0f;
        mapProps.push_back({ glm::vec3(x, groundY + h - 0.5f, z), glm::vec3(r, r*0.9f, r),
            glm::vec3(0.08f + (rand()%12)/100.0f, 0.31f + (rand()%16)/100.0f, 0.29f) });
    }
    for (int i = 0; i < 15; ++i) {
        float x = ((rand() % 1000) / 1000.0f - 0.5f) * 80.0f;
        float z = ((rand() % 1000) / 1000.0f - 0.5f) * 80.0f;
        if (std::sqrt(x*x + z*z) < 4.0f || PathDistance(x, z) < 3.5f) continue;
        float s = 0.6f + (rand() % 100) / 80.0f;
        addProp(x, z, glm::vec3(s, s*0.5f, s*0.8f), glm::vec3(0.29f, 0.36f, 0.37f));
    }
    for (int i = 0; i < 30; ++i) {
        float x = ((rand() % 1000) / 1000.0f - 0.5f) * 85.0f;
        float z = ((rand() % 1000) / 1000.0f - 0.5f) * 85.0f;
        if (PathDistance(x, z) < 3.0f) continue;
        float s = 0.4f + (rand() % 60) / 100.0f;
        addProp(x, z, glm::vec3(s, s*0.6f, s),
            glm::vec3(0.07f + (rand()%10)/100.0f, 0.29f + (rand()%15)/100.0f, 0.27f));
    }

    struct RuinDef { float x,z,w,h,d; };
    RuinDef ruins[] = {
        {-18,-15,4,5,3}, {-16,-15,1,3.5f,3}, {-20,-13,3.5f,2,0.6f},
        {20,-18,5,4.5f,4}, {23,-17,1.2f,6,1.2f}, {20,-14.5f,4.5f,1.5f,0.5f},
        {-10,-28,6,3,5}, {-7.5f,-27,0.8f,4.5f,0.8f}, {-13,-27,0.8f,2.5f,0.8f},
        {12,15,3.5f,3.5f,3}, {14.5f,16,1,5.5f,1},
    };
    for (const auto& r : ruins) {
        glm::vec3 c = r.h > 4.0f ? glm::vec3(0.20f,0.29f,0.31f)
            : (r.h < 2.0f ? glm::vec3(0.26f,0.34f,0.34f) : glm::vec3(0.31f,0.38f,0.38f));
        addProp(r.x, r.z, glm::vec3(r.w, r.h, r.d), c);
    }

    // 半水没した交通地区を構成する。高層建築、崩落した連絡橋、
    // 潮位ビーコンを同じワールド座標系に配置する。
    for (int i = 0; i < 14; ++i) {
        float angle = (float)i * 0.74f;
        float radius = 18.0f + (i % 4) * 7.0f;
        float x = std::cos(angle) * radius;
        float z = std::sin(angle) * radius - 6.0f;
        float h = 7.0f + (i % 5) * 3.2f;
        addProp(x, z, glm::vec3(2.4f + (i % 3), h, 2.1f + ((i + 1) % 3)),
            glm::vec3(0.12f + (i % 2) * 0.04f, 0.23f, 0.27f));
        mapProps.push_back({ glm::vec3(x, RawTerrainHeight(x, z) + h + 0.3f, z),
            glm::vec3(0.22f, 0.65f, 0.22f), glm::vec3(0.34f, 0.91f, 0.82f) });
    }
    for (int i = -5; i <= 5; ++i) {
        float x = i * 4.0f;
        float z = -8.0f + std::sin((float)i * 0.7f) * 1.5f;
        mapProps.push_back({ glm::vec3(x, RawTerrainHeight(x, z) + 1.1f, z),
            glm::vec3(1.8f, 0.18f, 1.0f), glm::vec3(0.25f, 0.34f, 0.35f) });
    }

    struct FenceDef { float x,z; int count; float angle; };
    FenceDef fences[] = { {-5,-22,5,0}, {10,-10,4,1.2f}, {-25,5,3,0.5f} };
    for (const auto& f : fences) {
        for (int i = 0; i < f.count; ++i) {
            float off = (float)i * 1.8f;
            float px = f.x + std::cos(f.angle) * off;
            float pz = f.z + std::sin(f.angle) * off;
            addProp(px, pz, glm::vec3(0.15f, 1.2f + (rand()%40)/100.0f, 0.15f), glm::vec3(0.38f,0.30f,0.18f));
            if (i < f.count - 1) {
                float rx = f.x + std::cos(f.angle)*(off+0.9f);
                float rz = f.z + std::sin(f.angle)*(off+0.9f);
                mapProps.push_back({ glm::vec3(rx, RawTerrainHeight(rx, rz) + 0.6f, rz),
                    glm::vec3(1.6f,0.1f,0.1f), glm::vec3(0.35f,0.28f,0.16f) });
            }
        }
    }

    for (int i = 0; i < 8; ++i) {
        float x = ((rand() % 1000) / 1000.0f - 0.5f) * 70.0f;
        float z = ((rand() % 1000) / 1000.0f - 0.5f) * 70.0f;
        if (PathDistance(x, z) < 3.5f || std::sqrt(x*x + z*z) < 6.0f) continue;
        float s = 0.7f + (rand() % 50) / 100.0f;
        addProp(x, z, glm::vec3(s, s, s), glm::vec3(0.45f,0.35f,0.20f));
    }

    srand((unsigned int)time(NULL));
}

void DrawSimpleMap(const glm::mat4& view, const glm::mat4& projection, const glm::vec3& cameraPos)
{
    const glm::vec3 sunDirection = glm::normalize(glm::vec3(-0.38f, 0.82f, 0.31f));
    const glm::mat4 lightView = glm::lookAt(
        glm::vec3(-28.0f, 46.0f, 25.0f), glm::vec3(0.0f, 2.0f, 1.0f), glm::vec3(0, 1, 0));
    const glm::mat4 lightProjection = glm::ortho(-38.0f, 38.0f, -38.0f, 38.0f, 1.0f, 105.0f);
    const glm::mat4 lightSpace = lightProjection * lightView;

    if (drownedCityLoaded && cityShadowFbo != 0) {
        GLint viewport[4];
        glGetIntegerv(GL_VIEWPORT, viewport);
        glViewport(0, 0, CITY_SHADOW_SIZE, CITY_SHADOW_SIZE);
        glBindFramebuffer(GL_FRAMEBUFFER, cityShadowFbo);
        glClear(GL_DEPTH_BUFFER_BIT);
        glEnable(GL_DEPTH_TEST);
        glEnable(GL_CULL_FACE);
        glCullFace(GL_FRONT);
        glUseProgram(cityShadowShader);
        const glm::mat4 identity(1.0f);
        glUniformMatrix4fv(glGetUniformLocation(cityShadowShader, "model"), 1, GL_FALSE, glm::value_ptr(identity));
        glUniformMatrix4fv(glGetUniformLocation(cityShadowShader, "lightSpaceMatrix"), 1, GL_FALSE, glm::value_ptr(lightSpace));
        drownedCityModel.draw();
        glCullFace(GL_BACK);
        glBindFramebuffer(GL_FRAMEBUFFER, 0);
        glViewport(viewport[0], viewport[1], viewport[2], viewport[3]);
    }

    GLuint shader = GetActorShader();
    glUseProgram(shader);
    glUniformMatrix4fv(glGetUniformLocation(shader, "view"), 1, GL_FALSE, glm::value_ptr(view));
    glUniformMatrix4fv(glGetUniformLocation(shader, "projection"), 1, GL_FALSE, glm::value_ptr(projection));
    glUniform3fv(glGetUniformLocation(shader, "cameraPos"), 1, glm::value_ptr(cameraPos));
    glUniform3fv(glGetUniformLocation(shader, "lightDir"), 1, glm::value_ptr(sunDirection));
    glUniform1i(glGetUniformLocation(shader, "outlineMode"), 0);
    glUniform1i(glGetUniformLocation(shader, "useTexture"), 0);
    glUniform1f(glGetUniformLocation(shader, "opacity"), 1.0f);
    glUniform1f(glGetUniformLocation(shader, "materialRoughness"), 0.74f);
    glUniform1f(glGetUniformLocation(shader, "materialMetallic"), 0.0f);
    glUniform3f(glGetUniformLocation(shader, "materialEmission"), 0.0f, 0.0f, 0.0f);
    glUniform1i(glGetUniformLocation(shader, "environmentMode"), 1);
    glUniform1i(glGetUniformLocation(shader, "useShadowMap"), drownedCityLoaded ? 1 : 0);
    glUniformMatrix4fv(glGetUniformLocation(shader, "lightSpaceMatrix"), 1, GL_FALSE, glm::value_ptr(lightSpace));
    glActiveTexture(GL_TEXTURE5);
    glBindTexture(GL_TEXTURE_2D, cityShadowDepth);
    glUniform1i(glGetUniformLocation(shader, "shadowMap"), 5);
    glActiveTexture(GL_TEXTURE0);

    if (drownedCityLoaded) {
        glm::mat4 cityModel(1.0f);
        glUniformMatrix4fv(glGetUniformLocation(shader, "model"), 1, GL_FALSE, glm::value_ptr(cityModel));
        glUniform1i(glGetUniformLocation(shader, "useVertexColor"), 1);
        glUniform1i(glGetUniformLocation(shader, "useTexture"), 1);
        drownedCityModel.draw();
    }

    glm::mat4 groundModel(1.0f);
    glUniformMatrix4fv(glGetUniformLocation(shader, "model"), 1, GL_FALSE, glm::value_ptr(groundModel));
    glUniform1i(glGetUniformLocation(shader, "useVertexColor"), 1);
    glBindVertexArray(mapGroundVao);
    glDrawElements(GL_TRIANGLES, groundIndexCount, GL_UNSIGNED_INT, nullptr);

    glUniform1i(glGetUniformLocation(shader, "useVertexColor"), 0);
    glBindVertexArray(mapPropVao);
    if (!drownedCityLoaded) {
        for (const MapProp& prop : mapProps) {
            glm::mat4 model(1.0f);
            model = glm::translate(model, prop.position);
            model = glm::scale(model, prop.scale);
            glUniformMatrix4fv(glGetUniformLocation(shader, "model"), 1, GL_FALSE, glm::value_ptr(model));
            glUniform3fv(glGetUniformLocation(shader, "actorColor"), 1, glm::value_ptr(prop.color));
            glDrawElements(GL_TRIANGLES, mapPropIndexCount, GL_UNSIGNED_INT, nullptr);
        }
    }
    glBindVertexArray(0);
}
