#include "Title.h"
#include "imgui.h"
#include "Scene.h"
#include <cmath>
#include <iostream>

extern bool g_StoryNeedsLoad;

namespace
{
    void DrawTitleBackdrop(ImDrawList* dl, const ImVec2& size)
    {
        dl->AddRectFilledMultiColor(
            ImVec2(0, 0), size,
            IM_COL32(3, 13, 23, 255), IM_COL32(5, 30, 43, 255),
            IM_COL32(5, 18, 30, 255), IM_COL32(1, 8, 17, 255));

        const float horizon = size.y * 0.58f;
        dl->AddRectFilledMultiColor(
            ImVec2(0, horizon), size,
            IM_COL32(6, 47, 59, 245), IM_COL32(8, 55, 66, 245),
            IM_COL32(1, 12, 25, 255), IM_COL32(1, 10, 20, 255));

        // Drowned skyline and its reflection.
        for (int i = 0; i < 18; ++i) {
            float x = size.x * (0.02f + i * 0.058f);
            float w = size.x * (0.025f + (i % 4) * 0.007f);
            float h = size.y * (0.10f + (i % 5) * 0.035f);
            dl->AddRectFilled(ImVec2(x, horizon - h), ImVec2(x + w, horizon),
                IM_COL32(5, 20, 29, 235));
            if (i % 3 == 0) {
                dl->AddRectFilled(ImVec2(x + w * 0.44f, horizon - h - size.y * 0.035f),
                    ImVec2(x + w * 0.56f, horizon - h), IM_COL32(7, 22, 31, 225));
            }
            dl->AddRectFilledMultiColor(
                ImVec2(x, horizon), ImVec2(x + w, horizon + h * 0.55f),
                IM_COL32(10, 38, 48, 105), IM_COL32(10, 38, 48, 105),
                IM_COL32(4, 15, 26, 0), IM_COL32(4, 15, 26, 0));
        }

        const float time = (float)ImGui::GetTime();
        for (int i = 0; i < 9; ++i) {
            float y = horizon + 18.0f + i * 19.0f + std::sin(time * 0.45f + i) * 3.0f;
            float half = size.x * (0.18f + i * 0.045f);
            dl->AddLine(ImVec2(size.x * 0.5f - half, y), ImVec2(size.x * 0.5f + half, y),
                IM_COL32(72, 206, 213, 28 + i * 4), 1.0f);
        }

        // 第一章で繰り返し使用する「ビーコンと月」の視覚モチーフを描画する。
        ImVec2 beacon(size.x * 0.78f, size.y * 0.25f);
        for (int r = 70; r > 8; r -= 8)
            dl->AddCircleFilled(beacon, (float)r, IM_COL32(92, 226, 218, 2 + (70 - r) / 2));
        dl->AddCircleFilled(beacon, 7.0f, IM_COL32(205, 255, 242, 235));
        dl->AddLine(beacon, ImVec2(beacon.x, horizon + 90.0f), IM_COL32(106, 232, 224, 75), 1.0f);
    }
}

void TitleUpdate()
{
    ImVec2 screen = ImGui::GetIO().DisplaySize;
    ImDrawList* bg = ImGui::GetBackgroundDrawList();
    DrawTitleBackdrop(bg, screen);

    ImGuiWindowFlags flags = ImGuiWindowFlags_NoDecoration | ImGuiWindowFlags_NoMove |
        ImGuiWindowFlags_NoSavedSettings | ImGuiWindowFlags_NoBackground;
    ImGui::SetNextWindowPos(ImVec2(0, 0), ImGuiCond_Always);
    ImGui::SetNextWindowSize(screen, ImGuiCond_Always);
    ImGui::Begin("##TideglassTitle", nullptr, flags);

    const float left = screen.x * 0.095f;
    ImGui::SetCursorPos(ImVec2(left, screen.y * 0.18f));
    ImGui::TextColored(ImVec4(0.70f, 0.96f, 0.95f, 1.0f), "T I D E G L A S S");
    ImGui::SetCursorPosX(left);
    ImGui::TextColored(ImVec4(0.92f, 0.97f, 1.0f, 1.0f), u8"潮鏡都市 ― 沈んだ空の残響");
    ImGui::SetCursorPos(ImVec2(left, screen.y * 0.34f));
    ImGui::TextColored(ImVec4(0.58f, 0.76f, 0.80f, 1.0f),
        u8"海面下に眠る記憶を回収し、都市の最後の夜を取り戻せ。");

    ImGui::SetCursorPos(ImVec2(left, screen.y * 0.56f));
    ImGui::PushStyleVar(ImGuiStyleVar_FrameRounding, 2.0f);
    ImGui::PushStyleVar(ImGuiStyleVar_FrameBorderSize, 1.0f);
    ImGui::PushStyleColor(ImGuiCol_Button, ImVec4(0.03f, 0.16f, 0.20f, 0.90f));
    ImGui::PushStyleColor(ImGuiCol_ButtonHovered, ImVec4(0.06f, 0.31f, 0.35f, 0.95f));
    ImGui::PushStyleColor(ImGuiCol_ButtonActive, ImVec4(0.12f, 0.43f, 0.45f, 1.0f));
    ImGui::PushStyleColor(ImGuiCol_Border, ImVec4(0.31f, 0.78f, 0.78f, 0.75f));

    if (ImGui::Button(u8"物語をはじめる", ImVec2(320, 54))) {
        currentScene = Scene::StoryEvent;
        g_StoryNeedsLoad = true;
    }
    ImGui::SetCursorPosX(left);
    if (ImGui::Button(u8"戦闘シミュレーション", ImVec2(320, 46)))
        currentScene = Scene::Battle;
    ImGui::SetCursorPosX(left);
    if (ImGui::Button(u8"終了", ImVec2(320, 42)))
        exit(0);

    ImGui::PopStyleColor(4);
    ImGui::PopStyleVar(2);

    ImGui::SetCursorPos(ImVec2(left, screen.y - 48.0f));
    ImGui::TextColored(ImVec4(0.38f, 0.58f, 0.62f, 1.0f),
        u8"WASD 移動  /  右ドラッグ カメラ  /  E 会話  /  F11 全画面");
    ImGui::End();
}
